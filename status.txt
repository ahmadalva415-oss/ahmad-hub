-- وظيفة السحب (Draggable) للقائمة
local function makeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true dragStart = input.Position startPos = gui.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

-- وظيفة إضافة الأزرار
local function addToggle(name, parent, callback)
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(1, -10, 0, 50)
    b.Text = name .. " [OFF]"
    b.BackgroundColor3 = Color3.fromRGB(40, 40, 45)
    b.TextColor3 = Color3.fromRGB(255, 255, 255)
    b.Parent = parent
    Instance.new("UICorner", b)
    local s = false
    b.MouseButton1Click:Connect(function()
        s = not s
        b.Text = name .. (s and " [ON]" or " [OFF]")
        b.BackgroundColor3 = s and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(40, 40, 45)
        callback(s)
    end)
end

-- وظيفة إضافة حقل إدخال للمسافة
local function addInput(name, default, parent, callback)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 60)
    frame.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    frame.Parent = parent
    Instance.new("UICorner", frame)

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, 0, 1, 0)
    label.Text = name
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.BackgroundTransparency = 1
    label.Parent = frame

    local input = Instance.new("TextBox")
    input.Size = UDim2.new(0.3, 0, 0.6, 0)
    input.Position = UDim2.new(0.65, 0, 0.2, 0)
    input.Text = tostring(default)
    input.BackgroundColor3 = Color3.fromRGB(50, 50, 55)
    input.TextColor3 = Color3.fromRGB(255, 255, 255)
    input.Parent = frame
    Instance.new("UICorner", input)

    input.FocusLost:Connect(function()
        local val = tonumber(input.Text)
        if val then callback(val) end
    end)
end

--- [ الأوامر ] ---
addToggle("سحب وتثبيت الجميع", SelfTab, function(v) dragAllEnabled = v end)

-- حلقة العمل
RunService.RenderStepped:Connect(function()
    if dragAllEnabled then
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            -- تحديث الموقع بناءً على القيم المدخلة في القائمة
            -- نستخدم -distanceForward لأن الاتجاه الأمامي في Roblox هو بالسالب على محور Z
            local targetCFrame = char.HumanoidRootPart.CFrame * CFrame.new(0, heightOffset, -distanceForward)
            
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    local hrp = p.Character.HumanoidRootPart
                    hrp.CFrame = targetCFrame
                    hrp.Velocity = Vector3.new(0, 0, 0)
                end
            end
        end
    end
end)
